document.getElementById('rechargeForm').addEventListener('submit', async function(e) {
  e.preventDefault();
  if (!currentUser) { alert('Connecte-toi d\'abord !'); return; }
  const type = document.getElementById('rechargeDrogue').value;
  const quantite = parseInt(document.getElementById('rechargeQuantite').value) || 0;
  const taux = parseInt(document.getElementById('rechargeTaux').value) || 0;
  const resultDiv = document.getElementById('rechargeResult');
  if (quantite <= 0) {
    resultDiv.innerHTML = '<span style="color:#FF5C5C;">❌ Entrez une quantité valide.</span>';
    return;
  }
  const submitBtn = this.querySelector('button[type="submit"]');
  submitBtn.disabled = true; submitBtn.textContent = '⏳ Envoi...';
  resultDiv.innerHTML = '<span style="color:#7CFFA0;">⏳ Rechargement en cours...</span>';

  try {
    // Vérifier si le type existe déjà
    const { data: existing, error: checkErr } = await supabase
      .from('stock_drogues')
      .select('type, quantite')
      .eq('type', type)
      .maybeSingle();
    if (checkErr) throw checkErr;
    if (existing) {
      const newQty = Number(existing.quantite) + quantite;
      const { error: updErr } = await supabase
        .from('stock_drogues')
        .update({ quantite: newQty })
        .eq('type', type);
      if (updErr) throw updErr;
    } else {
      const { error: insErr } = await supabase
        .from('stock_drogues')
        .insert([{ type: type, quantite: quantite }]);
      if (insErr) throw insErr;
    }
    // Ajouter à l'historique
    await supabase
      .from('historique')
      .insert([{ action: 'RECHARGEMENT STOCK', auteur: 'Admin', details: type + ' +' + quantite + ' (taux ' + taux + '%)', created_at: new Date().toISOString() }]);
    resultDiv.innerHTML = '<span style="color:#39FF6A;">✅ Stock rechargé : ' + quantite + ' pochons de ' + type + ' ajoutés.</span>';
    submitBtn.disabled = false; submitBtn.textContent = '💾 Recharger';
    chargerStockDrogues();
    setTimeout(() => fermerModal('rechargeModal'), 1500);
  } catch (err) {
    resultDiv.innerHTML = '<span style="color:#FF5C5C;">❌ Erreur : ' + err.message + '</span>';
    submitBtn.disabled = false; submitBtn.textContent = '💾 Recharger';
  }
});

document.getElementById('ticketForm').addEventListener('submit', async function(e) {
  e.preventDefault();
  if (!currentUser) { alert('Connecte-toi d\'abord !'); return; }
  const drogue = document.getElementById('ticketDrogue').value;
  const quantite = parseInt(document.getElementById('ticketQuantite').value) || 0;
  const taux = parseInt(document.getElementById('ticketTaux').value) || 0;
  const montant = parseInt(document.getElementById('ticketMontant').value) || 0;
  const resultDiv = document.getElementById('ticketResult');
  if (quantite <= 0 || montant <= 0) {
    resultDiv.innerHTML = '<span style="color:#FF5C5C;">❌ Entrez une quantité et un montant valides.</span>';
    return;
  }
  const submitBtn = this.querySelector('button[type="submit"]');
  submitBtn.disabled = true; submitBtn.textContent = '⏳ Envoi...';
  resultDiv.innerHTML = '<span style="color:#7CFFA0;">⏳ Vente en cours...</span>';

  const pseudo = currentUser.pseudo;
  const nom = pseudo;

  try {
    // 1) Vérifier le stock
    const { data: stockData, error: stockErr } = await supabase
      .from('stock_drogues')
      .select('quantite')
      .eq('type', drogue)
      .maybeSingle();
    if (stockErr) throw stockErr;
    if (!stockData) throw new Error('Type de drogue inconnu dans le stock.');
    const stockActuel = Number(stockData.quantite);
    if (stockActuel < quantite) {
      throw new Error('Stock insuffisant pour ' + drogue + '. Il reste ' + stockActuel + ' pochons.');
    }

    // 2) Déduire du stock
    const nouveauStock = stockActuel - quantite;
    const { error: updStock } = await supabase
      .from('stock_drogues')
      .update({ quantite: nouveauStock })
      .eq('type', drogue);
    if (updStock) throw updStock;

    // 3) Mettre à jour le membre
    const { data: memberData, error: memberErr } = await supabase
      .from('membres')
      .select('drogue, apayer')
      .eq('nom', nom)
      .maybeSingle();
    if (memberErr) throw memberErr;
    if (!memberData) throw new Error('Membre introuvable : ' + nom);
    const nouvelleDrogue = Number(memberData.drogue) + quantite;
    const nouvelAPayer = Number(memberData.apayer) + montant;
    const { error: updMember } = await supabase
      .from('membres')
      .update({ drogue: nouvelleDrogue, apayer: nouvelAPayer, quota: nouvelAPayer })
      .eq('nom', nom);
    if (updMember) throw updMember;

    // 4) Enregistrer la vente
    const { error: insVente } = await supabase
      .from('ventes_drogues')
      .insert([{ type: drogue, quantite: quantite, montant: montant, taux: taux, vendeur: nom, created_at: new Date().toISOString() }]);
    if (insVente) throw insVente;

    // 5) Historique
    await supabase
      .from('historique')
      .insert([{ action: 'VENTE DE DROGUE', auteur: nom, details: drogue + ' x' + quantite + ' = ' + montant + '$ (taux ' + taux + '%)', created_at: new Date().toISOString() }]);

    resultDiv.innerHTML = '<span style="color:#39FF6A;">✅ Vente déclarée !<br>Drogue : ' + drogue + '<br>Pochons vendus : ' + quantite + '<br>Montant : ' + montant + ' $<br>Taux : ' + taux + '%<br>Stock restant : ' + nouveauStock + ' pochons</span>';
    submitBtn.disabled = false; submitBtn.textContent = '💾 Valider la vente';
    chargerStockDrogues();
    chargerVentesStats();
    setTimeout(() => {
      chargerDonnees();
      fermerModal('ticketModal');
      updateLastUpdate();
      toast('📊 Tableaux actualisés avec succès');
    }, 800);
  } catch (err) {
    console.error('Erreur vente:', err);
    resultDiv.innerHTML = '<span style="color:#FF5C5C;">❌ Erreur : ' + err.message + '</span>';
    submitBtn.disabled = false; submitBtn.textContent = '💾 Valider la vente';
  }
});

// STATISTIQUES DES VENTES (50/70/90/99)
// ============================================================
async function chargerVentesStats() {
  try {
    const { data: ventes, error } = await supabase
      .from('ventes_drogues')
      .select('taux, quantite')
      .order('created_at', { ascending: false })
      .limit(20);
    if (error) throw error;
    const container = document.getElementById('tauxChart');
    if (!container) return;
    if (!ventes || ventes.length === 0) {
      container.innerHTML = '<div style="color:#8aa89a;text-align:center;width:100%;">Aucune vente enregistrée.</div>';
      return;
    }
    // Calcul des tranches
    const tranches = { '50%': 0, '70%': 0, '90%': 0, '99%': 0 };
    ventes.forEach(v => {
      const t = Number(v.taux);
      const q = Number(v.quantite);
      if (t >= 45 && t <= 54) tranches['50%'] += q;
      else if (t >= 65 && t <= 74) tranches['70%'] += q;
      else if (t >= 85 && t <= 94) tranches['90%'] += q;
      else if (t >= 95) tranches['99%'] += q;
    });
    const max = Math.max(1, ...Object.values(tranches));
    let html = '';
    const order = ['50%', '70%', '90%', '99%'];
    order.forEach(key => {
      const h = Math.max(6, Math.round((tranches[key] / max) * 100));
      html += `<div style="flex:1;display:flex;flex-direction:column;align-items:center;height:100%;justify-content:flex-end;">
        <div style="width:100%;max-width:30px;background:#39FF6A;border-radius:4px 4px 0 0;height:${h}px;opacity:0.8;"></div>
        <div style="font-size:8px;color:#8aa89a;margin-top:2px;">${tranches[key]}</div>
      </div>`;
    });
    container.innerHTML = html;

    // Tableau des dernières ventes
    const tbody = document.getElementById('ventesTableBody');
    if (!tbody) return;
    const { data: lastVentes } = await supabase
      .from('ventes_drogues')
      .select('*')
      .order('created_at', { ascending: false })
      .limit(10);
    if (!lastVentes || lastVentes.length === 0) {
      tbody.innerHTML = '<tr><td colspan="6" style="text-align:center;color:#8aa89a;">Aucune vente récente.</td></tr>';
      return;
    }
    tbody.innerHTML = lastVentes.map(v => {
      const date = new Date(v.created_at).toLocaleDateString('fr-FR') + ' ' + new Date(v.created_at).toLocaleTimeString('fr-FR', {hour:'2-digit',minute:'2-digit'});
      return `<tr><td>${date}</td><td>${v.vendeur}</td><td>${v.type}</td><td>${v.quantite}</td><td>${Number(v.montant).toLocaleString('fr-FR')} $</td><td style="font-weight:700;color:${v.taux >= 80 ? '#39FF6A' : v.taux >= 50 ? '#FFD86B' : '#FFA53C'};">${v.taux}%</td></tr>`;
    }).join('');
  } catch (e) {
    console.error('Erreur stats ventes:', e);
  }
}

async function chargerMissions() {
  try {
    const { data, error } = await supabase
      .from('missions')
      .select('*')
      .order('id', { ascending: true });
    if (error) throw error;
    allMissions = data || [];
    const grid = document.getElementById('missionsGrid');
    if (!data || data.length === 0) {
      grid.innerHTML = '<div style="text-align:center;color:#8aa89a;padding:40px;">Aucune mission.</div>';
      return;
    }
    grid.innerHTML = data.map((ms, idx) => {
      let badgeText = 'STANDARD', badgeClass = 'low';
      const quotaNum = parseFloat(String(ms.quota).replace(/[^0-9.-]+/g,'')) || 0;
      if (quotaNum > 100000) { badgeText = '🔥 HIGH PRIORITY'; badgeClass = 'high'; }
      else if (quotaNum > 50000) { badgeText = '⚡ MEDIUM RISK'; badgeClass = 'medium'; }
      return `<div class="mission-card"><div style="display:flex;justify-content:space-between;align-items:flex-start;margin-bottom:14px;"><div class="mission-title">${ms.titre||'MISSION '+(idx+1)}</div><span class="mission-badge ${badgeClass}">${badgeText}</span></div><div class="mission-detail-row"><span class="label">📍 OBJECTIF</span><span class="value">${ms.detail1||'—'}</span></div><div class="mission-detail-row"><span class="label">📊 RÉCOMPENSE</span><span class="value highlight">${ms.quota||'—'}</span></div><div class="mission-detail-row"><span class="label">📌 DÉTAIL</span><span class="value">${ms.detail2||'—'}</span></div></div>`;
    }).join('');
    const preview = document.getElementById('missionsDashPreview');
    if(data && data.length > 0) {
      preview.innerHTML = data.slice(0,2).map(ms => `<div style="padding:4px 0;border-bottom:1px solid #14251c;">• <strong>${ms.titre}</strong> <span style="color:#FFD86B;">${ms.quota}</span></div>`).join('');
      if(data.length > 2) preview.innerHTML += `<div style="padding:4px 0;color:#8aa89a;">Et ${data.length-2} autres...</div>`;
    } else {
      preview.innerHTML = 'Aucune mission active.';
    }
    if (isAdmin()) {
      const container = document.getElementById('missionsEditContainer');
      container.innerHTML = data.map((ms, idx) => {
        const num = idx+1;
        return `<div style="border-bottom:1px solid #1f5c3a;padding-bottom:12px;margin-bottom:12px;"><div style="color:#39FF6A;font-weight:700;margin-bottom:6px;">Mission ${num}</div><div class="modal-field"><label>Titre</label><input type="text" id="mission${num}_titre" value="${ms.titre||''}"></div><div class="modal-field"><label>Détail 1</label><input type="text" id="mission${num}_detail1" value="${ms.detail1||''}"></div><div class="modal-field"><label>Détail 2</label><input type="text" id="mission${num}_detail2" value="${ms.detail2||''}"></div><div class="modal-field"><label>Quota</label><input type="text" id="mission${num}_quota" value="${ms.quota||''}"></div><div class="modal-field"><label>Image URL</label><input type="text" id="mission${num}_image" value="${ms.imageURL||''}"></div></div>`;
      }).join('');
    }
  } catch (e) {
    console.error('Erreur missions:', e);
  }
}
async function chargerEtatMissions() {
  try {
    const { data, error } = await supabase
      .from('configuration')
      .select('valeur')
      .eq('propriete', 'MissionsActive')
      .maybeSingle();
    if (error) throw error;
    missionsActive = data ? data.valeur === true : true;
    const btn = document.getElementById('toggleMissionBtn');
    if(btn) {
      btn.textContent = missionsActive ? '🌿 Actif' : '🚫 Inactif';
      btn.style.borderColor = missionsActive ? '#39FF6A55' : '#FF5C5C55';
    }
    const grid = document.getElementById('missionsGrid');
    if(grid) grid.style.display = missionsActive ? 'grid' : 'none';
  } catch (e) {
    console.error('Erreur état missions:', e);
  }
}
async function toggleMissions() {
  if(!isAdmin()) return;
  const newState = !missionsActive;
  try {
    await supabase
      .from('configuration')
      .upsert({ propriete: 'MissionsActive', valeur: newState }, { onConflict: 'propriete' });
    chargerEtatMissions();
    toast("Missions " + (newState ? "activées" : "désactivées") + " !");
  } catch (err) { alert('Erreur : ' + err.message); }
}
document.getElementById('missionsForm').addEventListener('submit', async function(e) {
  e.preventDefault();
  if (!isAdmin()) return;
  const missions = [];
  const count = document.querySelectorAll('#missionsEditContainer > div').length;
  for (let i=1; i<=count; i++) {
    const titre = document.getElementById('mission'+i+'_titre').value.trim();
    const detail1 = document.getElementById('mission'+i+'_detail1').value.trim();
    const detail2 = document.getElementById('mission'+i+'_detail2').value.trim();
    const quota = document.getElementById('mission'+i+'_quota').value.trim();
    const imageURL = document.getElementById('mission'+i+'_image').value.trim();
    missions.push({ titre, detail1, detail2, quota, imageURL });

  try {

      await supabase.from('missions').insert([m]);
    }
    alert('Missions sauvegardées.');
    chargerMissions();
    updateLastUpdate();
  } catch (err) { alert('Erreur : ' + err.message); }
});

function initChat(pseudo, role) {
  if (chatInitialized) return;
  const messagesContainer = document.getElementById('chatMessages');
  const input = document.getElementById('chatInput');
  const sendBtn = document.getElementById('chatSendBtn');
  if (!messagesContainer || !input || !sendBtn) return;
  input.disabled = false; sendBtn.disabled = false;
  input.placeholder = 'Écris ton message...';
  messagesContainer.innerHTML = '<div style="text-align:center;color:#555;font-size:13px;margin:auto;">Chargement...</div>';
  async function loadMessages() {
    const { data, error } = await supabase
      .from('messages')
      .select('*')
      .order('created_at', { ascending: true })
      .limit(100);
    if (error) { console.error('Erreur chargement:', error); messagesContainer.innerHTML = '<div style="color:#FF5C5C;text-align:center;">Erreur de chargement.</div>'; return; }
    messagesContainer.innerHTML = '';
    data.forEach(msg => appendMessage(msg));
    messagesContainer.scrollTop = messagesContainer.scrollHeight;
  }
  async function checkAndResetChat() {
    const lastReset = localStorage.getItem('chatLastReset');
    const now = Date.now();
    if (!lastReset || (now - parseInt(lastReset)) > 24 * 60 * 60 * 1000) {
      const oneDayAgo = new Date(now - 24 * 60 * 60 * 1000).toISOString();
      await supabase.from('messages').delete().lt('created_at', oneDayAgo);
      localStorage.setItem('chatLastReset', now.toString());
      console.log('✅ Chat reset automatique effectué (24h).');
    }
  }
  function appendMessage(msg) {
    const isSelf = msg.user_name === pseudo;
    const div = document.createElement('div');
    div.className = 'message' + (isSelf ? ' self' : '');
    const adminClass = msg.user_role === 'admin' ? 'admin' : '';
    div.innerHTML = `<div class="msg-header"><span class="name ${adminClass}">${msg.user_name}</span><span>${new Date(msg.created_at).toLocaleTimeString('fr-FR')}</span></div><div class="msg-content">${msg.content}</div>`;
    messagesContainer.appendChild(div);
    messagesContainer.scrollTop = messagesContainer.scrollHeight;
  }
  async function sendMessage() {
    const content = input.value.trim();
    if (!content) return;
    input.value = ''; input.disabled = true; sendBtn.disabled = true;
    const { error } = await supabase
      .from('messages')
      .insert([{ user_name: pseudo, user_role: role || 'membre', content, created_at: new Date().toISOString() }]);
    if (error) { console.error('Erreur envoi:', error); alert('Erreur lors de l\'envoi du message.'); }
    input.disabled = false; sendBtn.disabled = false; input.focus();
  }
  // Écouter les nouveaux messages en temps réel via Supabase Realtime
  supabase.channel('public:messages')
    .on('postgres_changes', { event: 'INSERT', schema: 'public', table: 'messages' }, (payload) => {
      appendMessage(payload.new);
    })
    .subscribe();
  sendBtn.addEventListener('click', sendMessage);
  input.addEventListener('keydown', (e) => { if (e.key === 'Enter') sendMessage(); });
  checkAndResetChat();
  loadMessages();
  chatInitialized = true;
}
function ouvrirChat() {
  document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
  const chatPage = document.getElementById('page-chat');
  if (chatPage) chatPage.classList.add('active');
  document.querySelectorAll('.tab').forEach(t => t.classList.remove('active'));
  document.querySelectorAll('.chat-widget').forEach(w => w.classList.add('active'));
  if (currentUser) {
    const pseudo = currentUser.pseudo;
    if (pseudo && !chatInitialized) {
      initChat(pseudo, currentUser.role || 'membre');
    }
  }
}

function genererBoutonsLoc() {
  const container = document.getElementById('locButtonsContainer');
  if (!container) return;
  container.innerHTML = locCategories.map(cat => {
    return `
      <button class="btn-loc" data-id="${cat.id}">
        <img src="${cat.image}" alt="${cat.label}">
        <div class="loc-label">${cat.label}</div>
      </button>
    `;
  }).join('');
  container.querySelectorAll('.btn-loc').forEach(btn => {
    btn.addEventListener('click', function() {
      const id = this.dataset.id;
      afficherImageLoc(id);
    });
  });
}
function afficherImageLoc(categoryId) {
  const cat = locCategories.find(c => c.id === categoryId);
  if (!cat) return;
  const placeholder = document.getElementById('locImagePlaceholder');
  const loader = document.getElementById('locImageLoader');
  const img = document.getElementById('locImageDisplay');
  placeholder.style.display = 'none';
  loader.style.display = 'block';
  img.style.display = 'none';
  const imgUrl = cat.image;
  const tempImg = new Image();
  tempImg.onload = function() {
    loader.style.display = 'none';
    img.src = imgUrl;
    img.style.display = 'block';
  };
  tempImg.onerror = function() {
    loader.style.display = 'none';
    img.style.display = 'none';
    placeholder.style.display = 'block';
    placeholder.textContent = '❌ Image introuvable pour ' + cat.label;
  };
  tempImg.src = imgUrl;
}
